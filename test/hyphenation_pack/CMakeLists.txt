cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(hyphenation_pack_tests LANGUAGES CXX)
  set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  enable_testing()
endif()
add_executable(HyphenationPackTest HyphenationPackTest.cpp ${REPO_ROOT}/lib/HyphenationPack/HyphenationPack.cpp)
target_include_directories(HyphenationPackTest PRIVATE ${REPO_ROOT}/lib/HyphenationPack ${REPO_ROOT}/lib/Memory)
target_compile_features(HyphenationPackTest PRIVATE cxx_std_20)
target_compile_options(HyphenationPackTest PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME HyphenationPackTest COMMAND HyphenationPackTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME HyphenationPackageTest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/package_test.py
  ${REPO_ROOT} $<TARGET_FILE:HyphenationPackTest>)
